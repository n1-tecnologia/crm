// apps/api/src/importacao/clickup/manifesto.ts
export interface EntradaDoManifesto { readonly caminho: string; readonly bytes: number; readonly sha256: string } // caminho relativo à pasta exportada, separador "/"; anexos: "<listId>/attachments/<taskId>/<attId>" (sem o nome original)
export interface Manifesto {
  readonly versao: 1;
  readonly exportadoEm: string;                       // ISO — vem de opcoes
  readonly exportador: { readonly commit: string };   // git rev-parse HEAD (o CLI passa)
  readonly amostra: boolean;                          // de execucao.json quando existe; senão false
  readonly timeInStatusDisponivel: boolean | "nao-verificado";
  readonly listas: readonly ContagensDaLista[];       // recontadas do disco (contarRecursos), com maiorDateUpdated
  readonly usuariosVistos: readonly { readonly id: string; readonly ocorrencias: number }[];  // SEM nome (de usuarios-vistos.json)
  readonly avisos: readonly string[];                 // de execucao.json quando existe
  readonly arquivos: readonly EntradaDoManifesto[];   // ordenado por caminho; exclui exportacao.log, execucao.json e *.tmp
  readonly hashAgregado: string;                      // sha256 das linhas "<sha256>  <caminho>\n" na ordem de `arquivos`
}
export async function hashDoArquivo(caminho: string): Promise<{ readonly sha256: string; readonly bytes: number }>;   // stream
export async function listarArquivos(pastaExportada: string): Promise<readonly EntradaDoManifesto[]>;
export async function contarRecursos(pastaExportada: string): Promise<readonly ContagensDaLista[]>;              // uma entrada por subpasta <listId>
export async function gerarManifesto(opcoes: { readonly pastaExportada: string; readonly exportadoEm: string; readonly commit: string }): Promise<Manifesto>;
export function manifestoEmMarkdown(manifesto: Manifesto): string;            // resumo legível; NÃO lista arquivo a arquivo (isso é o JSON)
export async function escreverManifesto(manifesto: Manifesto, pastaDeSaida: string): Promise<{ readonly json: string; readonly md: string }>; // grava MANIFESTO.json (JSON.stringify(m, null, 2) + "\n") e MANIFESTO.md; devolve os caminhos
