// apps/api/src/importacao/clickup/cliente.ts
export const BASE_URL = "https://api.clickup.com/api/v2";
export const TETO_DE_TENTATIVAS = 5;
export const HOSTS_PERMITIDOS = [".clickup.com", ".clickup-attachments.com"]; // sufixos; o token só viaja para estes
export interface RespostaClickUp { readonly status: number; readonly corpoCru: string; readonly json: unknown; readonly cabecalhos: Headers }
export interface ClienteClickUp {
  get(caminho: string, query?: Record<string, string>): Promise<RespostaClickUp>;   // caminho relativo a BASE_URL, ex.: "/list/901109104938/task"
  baixar(url: string): Promise<{ readonly bytes: Uint8Array; readonly contentType: string | null }>; // anexos; recusa host fora da allowlist
}
export interface OpcoesDoCliente {
  readonly token: string;
  readonly fetch?: typeof fetch;                 // injetável nos testes
  readonly dormir?: (ms: number) => Promise<void>; // injetável nos testes
  readonly agora?: () => number;                 // injetável nos testes (Date.now)
}
export class ErroClickUp extends Error { readonly status: number; readonly caminho: string }  // mensagem = status + caminho + trecho do corpo (≤ 300 chars); NUNCA headers
export function hostPermitido(url: string): boolean;
export function criarClienteClickUp(opcoes: OpcoesDoCliente): ClienteClickUp;

// apps/api/src/importacao/clickup/paginacao.ts
export interface PaginaPorNumero { readonly page: number; readonly corpoCru: string; readonly itens: readonly unknown[]; readonly ultima: boolean }
export interface CursorDeComentarios { readonly start: string; readonly start_id: string }
export interface PaginaPorCursor { readonly indice: number; readonly corpoCru: string; readonly itens: readonly unknown[]; readonly cursor?: CursorDeComentarios }
export const TETO_DE_PAGINAS = 1000;
export const COMENTARIOS_POR_PAGINA = 25;
export function paginarPorPagina(
  buscar: (page: number) => Promise<RespostaClickUp>,
  extrair: (json: unknown) => { readonly itens: readonly unknown[]; readonly ultima: boolean },   // ex.: tasks + last_page
): AsyncGenerator<PaginaPorNumero>;
export function paginarPorCursor(
  buscar: (cursor?: CursorDeComentarios) => Promise<RespostaClickUp>,
  extrair: (json: unknown) => readonly { readonly id: string; readonly date: string }[],           // ex.: comments
): AsyncGenerator<PaginaPorCursor>;
</interfaces>
