export type RoleIdDeSistema = "role_prop" | "role_adm" | "role_mem" | "role_conv"; // os 4 que buildGestaoBootstrap() cria
export type PapelNoProcesso = "financeiro" | "aprovador" | "revisor";
export interface EntradaDeParaUsuario {
  readonly clickupUserId: string;        // ex.: "81591229"
  readonly nome: string;                 // username como aparece no ClickUp
  readonly email: string;                // identidade — vira Membro por inviteMember (D-09); minúsculas
  readonly roleIdEsperado: RoleIdDeSistema;
  readonly papelNoProcesso: PapelNoProcesso;
  readonly memberIdConhecido?: string;   // só quando já existe Membro na plataforma (81581816 → "mem_maryane"); a importação NÃO chama inviteMember
  readonly fonteDoEmail: "clickup-api" | "dono";
}
export const DE_PARA_USUARIOS: readonly EntradaDeParaUsuario[];               // exatamente 6
export function entradaPorClickupId(id: string): EntradaDeParaUsuario | undefined;

export type TenantDoAutomation = "tria" | "marcos-paulo" | "family-office";
export type TipoDeVariavelClickUp =
  | "token" | "id-de-team" | "id-de-lista" | "id-de-usuario" | "ids-de-usuario"
  | "nome-de-usuario" | "mencao" | "status" | "tag" | "opcao-de-campo" | "nome-de-campo" | "limite-de-valor";
export interface VariavelClickUp {
  readonly nome: string;                          // ex.: "CLICKUP_RESPONSAVEL_FINQ_ID"
  readonly tenant: TenantDoAutomation;            // pelo prefixo: sem prefixo = tria, MP_ = marcos-paulo, FO_ = family-office
  readonly tipo: TipoDeVariavelClickUp;
  readonly uso: string;                           // pt-BR: arquivo e função do automation/src que a lê, para quê
  readonly clickupUserId?: string;                // tipo "id-de-usuario", quando o id é derivável dos docs (inventário §3/§4.1) ou do código dos tenants
  readonly clickupUserIds?: readonly string[];    // tipo "ids-de-usuario"
  readonly valorConhecido?: string;               // só ids de Lista/Team/status/tag documentados no inventário — NUNCA em "token"
  readonly entraNoDePara: boolean;                // true sse (ids-)de-usuario e o id está em DE_PARA_USUARIOS
}
