```typescript
export interface EstadoDoServidor { readonly data: DataState; readonly revisao: number; readonly ordens: Ordens; readonly memberId: Id }
export type RespostaDeGravacao =
  | { readonly tipo: "diferenca"; readonly resultado: OperationResult<unknown>; readonly diferenca: Diferenca; readonly revisao: number; readonly revisaoAnterior: number }
  | { readonly tipo: "estado"; readonly resultado: OperationResult<unknown>; readonly estado: EstadoDoServidor }   // memória: a cópia vira o estado
  | { readonly tipo: "conflito"; readonly erro: string }                                                            // 409 → re-hidratar
  | { readonly tipo: "falha"; readonly erro: string; readonly sessaoExpirada?: boolean };                          // rede/401/403/500
export interface PedidoDeGravacao { readonly chamadas: readonly ChamadaDeOperacao[]; readonly revisao: number; readonly copia: DataState }
export interface Transporte {
  gravar(pedido: PedidoDeGravacao): Promise<RespostaDeGravacao>;
  hidratar(): Promise<EstadoDoServidor | { readonly tipo: "falha"; readonly erro: string }>;
}
export function transporteHttp(opcoes: { readonly apiUrl: string; readonly obterToken: () => Promise<string | undefined> }): Transporte;
export function transporteEmMemoria(): Transporte;   // gravar → { tipo: "estado", estado: { data: copia, ... } }; hidratar → falha
```
